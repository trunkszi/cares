cmake_minimum_required(VERSION 4.0)
include(cmake/CaresToolchain.cmake)
project(cares VERSION 0.1.0 DESCRIPTION "C++26 runtime-free Proactor DNS framework" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(CARES_BUILD_TESTS "Build deterministic tests" ${PROJECT_IS_TOP_LEVEL})
option(CARES_BUILD_EXAMPLES "Build examples" ${CARES_BUILD_TESTS})
option(CARES_BUILD_BENCHES "Build benchmarks" OFF)
option(CARES_BUILD_TOOLS "Build native probes" ${CARES_BUILD_TESTS})
option(CARES_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
option(CARES_INSTALL_PACKAGE "Install cares package" ON)
find_program(CARES_CCACHE ccache)
if(CARES_CCACHE)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${CARES_CCACHE}")
endif()
include(CaresCompilerOptions)
include(CaresReflection)
include(CaresSanitizers)
include(CaresTargets)
include(CaresCompileFail)
include(CaresCodegenGates)
cares_detect_reflection()
cares_create_libraries()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/CMakeLists.txt")
  add_subdirectory(src)
endif()
if(CARES_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(CARES_BUILD_EXAMPLES AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/CMakeLists.txt")
  add_subdirectory(examples)
endif()
if(CARES_BUILD_BENCHES AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/benches/CMakeLists.txt")
  add_subdirectory(benches)
endif()
if(CARES_INSTALL_PACKAGE)
  install(TARGETS cares_error cares_types cares_record cares_query_table cares_channel cares_resolver cares EXPORT cares-targets)
  install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}" PATTERN "probes" EXCLUDE)
  install(EXPORT cares-targets NAMESPACE cares:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/cares")
  configure_package_config_file(cmake/cares-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/cares-config.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/cares")
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/cares-config-version.cmake" VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMajorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/cares-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/cares-config-version.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/cares")
endif()
