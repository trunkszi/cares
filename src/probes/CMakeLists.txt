cmake_minimum_required(VERSION 4.0)
add_library(cares_probes STATIC cares_probes.cc)
add_library(cares::probes ALIAS cares_probes)
cares_configure_internal(cares_probes NO_SANITIZERS)
set_target_properties(cares_probes PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
target_compile_options(cares_probes PRIVATE -O3 -fno-lto)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(cares_probes PRIVATE -fno-profile-instr-generate -fno-coverage-mapping)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(cares_probes PRIVATE -fno-profile-arcs -fno-test-coverage -fno-profile-generate)
endif()
